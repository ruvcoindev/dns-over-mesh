package main import ( "log" "dns-over-mesh/mesh" "dns-over-mesh/dns" "dns-over-mesh/web" "dns-over-mesh/utils" ) func main() { // Инициализация логирования if err := utils.InitLogger("dns-over-mesh.log"); err != nil { log.Fatalf("Failed to initialize logger: %v", err) } utils.LogInfo("Logger initialized") // Инициализация MESH-сети meshNetwork := mesh.NewMeshNetwork() if err := meshNetwork.Start(); err != nil { utils.LogError(err) log.Fatalf("Failed to start mesh network: %v", err) } utils.LogInfo("Mesh network started") // Инициализация DNS-сервера dnsServer := dns.NewDNSServer(meshNetwork) if err := dnsServer.Start(); err != nil { utils.LogError(err) log.Fatalf("Failed to start DNS server: %v", err) } utils.LogInfo("DNS server started") // Запуск веб-сервера для мониторинга webServer := web.NewWebServer(dnsServer) if err := webServer.Start(); err != nil { utils.LogError(err) log.Fatalf("Failed to start web server: %v", err) } utils.LogInfo("Web server started") }
